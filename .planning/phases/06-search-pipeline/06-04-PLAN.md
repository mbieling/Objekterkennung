```typescript
// src/app/api/search/route.test.ts
// Tests für POST /api/search
// SEARCH-03: Gerankete Treffer mit similarity-Score
// SEARCH-04: Threshold-Filter konfigurierbar
// SEARCH-05: Limit-Parameter konfigurierbar

import { describe, it, expect, vi, beforeEach } from 'vitest'
import { type NextRequest } from 'next/server'

vi.mock('@/lib/db', () => ({
  db: vi.fn(),
}))
vi.mock('@/lib/s3', () => ({
  s3: { send: vi.fn() },
  BUCKET_THUMBNAILS: 'mock-bucket-thumbnails',
}))

const mockFetch = vi.fn()
global.fetch = mockFetch

import { db } from '@/lib/db'
import { s3 } from '@/lib/s3'
const mockDb = vi.mocked(db)
const mockS3Send = vi.mocked(s3.send)

// Hilfsfunktion: minimales JPEG-Bild-File (valider MIME-Type)
function makeImageRequest(url = 'http://localhost/api/search'): Request {
  const formData = new FormData()
  formData.append('image', new File([new Uint8Array([1, 2, 3])], 'foto.jpg', { type: 'image/jpeg' }))
  return new Request(url, { method: 'POST', body: formData })
}

// Hilfsfunktion: DB-Row für Treffer
function makeDbRow(similarity = '0.85') {
  return {
    id: '123e4567-e89b-12d3-a456-426614174000',
    name: 'Flansch A',
    part_number: 'FL-001',
    project: null,
    status: 'ready',
    similarity,
    created_at: '2026-01-01T00:00:00.000Z',
  }
}

describe('POST /api/search', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    process.env.WORKER_URL = 'http://localhost:8000'
    // S3 send (PutObjectCommand + DeleteObjectCommand) immer erfolgreich
    mockS3Send.mockResolvedValue({})
    // Worker gibt valides 768-dim Embedding zurück
    mockFetch.mockResolvedValue({
      ok: true,
      json: async () => ({ embedding: new Array(768).fill(0.1) }),
    })
  })

  // SEARCH-03: HTTP 200 mit geranketen Treffern
  it('gibt HTTP 200 mit geranketen Treffern zurück wenn Treffer vorhanden (SEARCH-03)', async () => {
    mockDb.mockResolvedValueOnce([makeDbRow('0.85')])

    const { POST } = await import('./route')
    const response = await POST(makeImageRequest() as NextRequest)
    const data = await response.json()

    expect(response.status).toBe(200)
    expect(data.results).toHaveLength(1)
    expect(data.results[0].similarity).toBe(0.85)
    expect(typeof data.results[0].similarity).toBe('number')
    expect(data.query.threshold).toBe(0.7)
    expect(data.query.limit).toBe(10)
    expect(data.query.results_count).toBe(1)
  })

  // SEARCH-03: Keine Treffer → leeres Array
  it('gibt HTTP 200 mit leerem results-Array zurück wenn keine Treffer gefunden (SEARCH-03)', async () => {
    mockDb.mockResolvedValueOnce([])

    const { POST } = await import('./route')
    const response = await POST(makeImageRequest() as NextRequest)
    const data = await response.json()

    expect(response.status).toBe(200)
    expect(data.results).toHaveLength(0)
    expect(data.query.results_count).toBe(0)
  })

  // SEARCH-03: Worker nicht erreichbar → HTTP 502
  it('gibt HTTP 502 zurück wenn Worker nicht erreichbar ist (SEARCH-03)', async () => {
    mockFetch.mockRejectedValueOnce(new Error('ECONNREFUSED'))

    const { POST } = await import('./route')
    const response = await POST(makeImageRequest() as NextRequest)

    expect(response.status).toBe(502)
    // S3 Cleanup muss auch bei Worker-Fehler aufgerufen worden sein
    // PutObjectCommand (Upload) + DeleteObjectCommand (Cleanup) = mindestens 2 send-Calls
    expect(mockS3Send).toHaveBeenCalledTimes(2)
  })

  // SEARCH-03: Worker gibt Fehler zurück → HTTP 502
  it('gibt HTTP 502 zurück wenn Worker einen Fehler zurückgibt (SEARCH-03)', async () => {
    mockFetch.mockResolvedValueOnce({ ok: false, status: 500 })

    const { POST } = await import('./route')
    const response = await POST(makeImageRequest() as NextRequest)

    expect(response.status).toBe(502)
    // S3 Cleanup muss auch bei Worker-Fehler-Response aufgerufen worden sein
    expect(mockS3Send).toHaveBeenCalledTimes(2)
  })

  // SEARCH-04: Threshold-Filter
  it('schließt Treffer unter dem Schwellwert aus — DB erhält threshold-Wert aus Query-Param (SEARCH-04)', async () => {
    mockDb.mockResolvedValueOnce([makeDbRow('0.75')])

    const { POST } = await import('./route')
    const url = 'http://localhost/api/search?threshold=0.8'
    const response = await POST(makeImageRequest(url) as NextRequest)
    const data = await response.json()

    expect(response.status).toBe(200)
    // Der threshold-Wert wurde korrekt aus dem Query-Param gelesen
    expect(data.query.threshold).toBe(0.8)
    // DB-Query wurde mit threshold=0.8 aufgerufen (der Mock liefert was er liefert)
    expect(mockDb).toHaveBeenCalledOnce()
  })

  // SEARCH-04: Ungültiger threshold → HTTP 400
  it('gibt HTTP 400 zurück bei ungültigem threshold-Parameter (SEARCH-04)', async () => {
    const { POST } = await import('./route')
    const response = await POST(makeImageRequest('http://localhost/api/search?threshold=1.5') as NextRequest)

    expect(response.status).toBe(400)
    // Kein S3-Upload, kein Worker-Call bei ungültigen Query-Params
    expect(mockS3Send).not.toHaveBeenCalled()
    expect(mockFetch).not.toHaveBeenCalled()
  })

  // SEARCH-05: Limit-Parameter
  it('begrenzt die Anzahl der Treffer per limit-Parameter (SEARCH-05)', async () => {
    mockDb.mockResolvedValueOnce([makeDbRow('0.90'), makeDbRow('0.80'), makeDbRow('0.70')])

    const { POST } = await import('./route')
    const url = 'http://localhost/api/search?limit=3'
    const response = await POST(makeImageRequest(url) as NextRequest)
    const data = await response.json()

    expect(response.status).toBe(200)
    expect(data.query.limit).toBe(3)
    expect(data.results).toHaveLength(3)
    expect(data.query.results_count).toBe(3)
  })

  // SEARCH-05: Ungültiges limit → HTTP 400
  it('gibt HTTP 400 zurück bei ungültigem limit-Parameter (SEARCH-05)', async () => {
    const { POST } = await import('./route')
    const response = await POST(makeImageRequest('http://localhost/api/search?limit=0') as NextRequest)

    expect(response.status).toBe(400)
    expect(mockS3Send).not.toHaveBeenCalled()
  })

  // Eingabe-Validierung: kein image-Feld
  it('gibt HTTP 400 zurück wenn kein image-Feld im FormData vorhanden (SEARCH-03)', async () => {
    const { POST } = await import('./route')
    const formData = new FormData()
    // Kein 'image'-Feld hinzufügen
    const request = new Request('http://localhost/api/search', { method: 'POST', body: formData })

    const response = await POST(request as NextRequest)

    expect(response.status).toBe(400)
    expect(mockS3Send).not.toHaveBeenCalled()
    expect(mockFetch).not.toHaveBeenCalled()
  })
})
```
