   ```sql
   create or replace function update_updated_at()
   returns trigger
   language plpgsql
   security invoker
   set search_path = ''
   as $$
   begin
     new.updated_at := now();
     return new;
   end;
   $$;

   create trigger update_parts_updated_at
   before update on parts
   for each row
   execute function update_updated_at();
   ```
